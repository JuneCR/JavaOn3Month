package com.june.ioc_05;
/*
* 制造JavaBean的工厂bean对象
*
* 步骤 1.实现FactoryBean接口<返回值泛型>
* */

import org.springframework.beans.factory.FactoryBean;

public class JavaBeanFactoryBean implements FactoryBean<JavaBean> {
    private String value;
    public void setValue(String value) {
        this.value = value;
    }
    @Override
    public JavaBean getObject() throws Exception {
        //使用你自己的方式实例化对象就可以了
        JavaBean javaBean = new JavaBean();
        javaBean.setName(value);
        return javaBean;
    }

    @Override
    public Class<?> getObjectType() {
        return JavaBean.class;
    }
}
FactoryBean使用场景
		a.代理类的创建
		b.第三方框架的整合
		c.复杂对象实例化等
		
		FactoryBean接口是SpringIoc容器实例化逻辑的可插拔性点
		用于配置复杂的Bean对象，可以将创建存储在FactoryBean的getObject方法
		/*FactoryBean 就是利用对象进行封装到方法来进行造对象*/
----------------------------------------------------------------------------
<!--
    id ->getObject 方法返回对象的标识
    Class ->factoryBean标准化工厂类
-->
    <bean id="javaBean" class="com.june.ioc_05.JavaBeanFactoryBean">
        <!--   此位置的属性:JavaBean工厂类配置 而不是getObject方法-->
        <property name="value" value="二狗子"/>
    </bean>

通常在xml配置方式中 想要声明类的信息需要<bean id="" class="类在的位置"/>

使用@Controller @Service @Repository 时默认id = 类名首字母小写

使用@Autowired 依赖注入其他组件
	在自动装配的过程中国会出现三种情况
	1.ioc容器种有一个UserSerivce对应的实现类
		直接使用
	2.ioc容器中没有对应的实现类
		使用注解时至少要有一个bean组件 否则报错
		解决方案: @Autowired(required = flase)能装就装,后期开发中要进行配置
	3.ioc容器中有多个组件(在di注入时有多个接口实现类不知道装哪个了)
		解决方案1:指定名
		@Autowired
		private UserService [userService]在这个位置指定你要装配的组件(注 首字母小写)
		解决方案2:使用@Qualifier(value = "userServiceImpl")
		使用该注解获取bean-id,不能单独使用要配合AutoWired
		[可以注解使用 @Resource(name = "userSerivceImpl")指定 bean-id]
@Value 用于给属性直接赋值,通常用于读取配置文件信息 格式:@Value($key:vlaue)

import org.springframework.beans.factory.config.ConfigurableBeanFactory;
import org.springframework.context.annotation.Scope;
import org.springframework.stereotype.Component;

import javax.annotation.PostConstruct;
import javax.annotation.PreDestroy;

@Scope(scopeName = ConfigurableBeanFactory.SCOPE_SINGLETON) //单例默认值
// ConfigurableBeanFactory.SCOPE_SINGLETON 多例
@Component
public class javaBean {


    //周期方法
    @PostConstruct
    public void init(){
        System.out.println("初始化");
    }
    @PreDestroy
    public void destroy(){
        System.out.println("销毁");
    }
}

package com.june.javaConfiguration;

import com.alibaba.druid.pool.DruidDataSource;
import org.springframework.beans.factory.annotation.Configurable;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.beans.factory.config.ConfigurableBeanFactory;
import org.springframework.context.annotation.*;
import org.springframework.jdbc.core.JdbcTemplate;

import javax.sql.DataSource;

//声明配置类
@Configuration
//包扫描
@ComponentScan({"com.june.ioc_01","com.june.ioc_02"})
//引入外部配置文件
@PropertySource("classpath:jdbc.properties")
public class JavaConfiguration {
    @Value("${url}")
    private String url;
    @Value("${driver}")
    private String driver;
    @Value("${username}")
    private String username;
    @Value("${password}")
    private String password;
//引入第三方组件
    /*
    * 问题1:beanName的问题
    *       默认:方法名
    *       指定： name/value属性起名字 覆盖方法名
    * 问题2:周期方法如何指定
    *       原有注解方案:PostConstruct+PreDestroy 注解指定
    *       bean属性指定 initMethod / destroyMethod 指定
    * 问题3:作用域
    *       和以前还是一样的
    * 问题4:如何引用其他的ioc主键
    * */
    @Scope(scopeName = ConfigurableBeanFactory.SCOPE_SINGLETON)
    @Bean(name="ergouzi",initMethod = "",destroyMethod = "")
    public DruidDataSource dataSource(){
        //实现具体的实例化过期
        DruidDataSource druidDataSource = new DruidDataSource();
        druidDataSource.setUrl(url);
        druidDataSource.setDriverClassName(driver);
        druidDataSource.setUsername(username);
        druidDataSource.setPassword(password);

        return druidDataSource;
    }
    @Bean
    public JdbcTemplate jdbcTemplate(){
        JdbcTemplate jdbcTemplate = new JdbcTemplate();
        //需要DataSource,需要oc容器的其他组件
        //方案1:如果其他组件也是@Bean方法可以直接调用
        jdbcTemplate.setDataSource(dataSource());
        return jdbcTemplate;
    }
    @Bean
    public JdbcTemplate jdbcTemplate1(DataSource dataSource){
        JdbcTemplate jdbcTemplate = new JdbcTemplate();
        //需要DataSource,需要oc容器的其他组件
        //方案2:形参列表声明想要的组件类，可以是一个也可以多个！ ioc会自动注入
        //形参变量注入，必须要有 DataSouce组件 ,如果没有就抛异常
        //如果有多个:可以使用形参名 DataSource <dataSource>来指定哪个DataSource
        jdbcTemplate.setDataSource(dataSource());
        return jdbcTemplate;
    }
}
